########## OPEN STORAGE ##########
automacro OpenStorage {
    timeout 10
    CharCurrentWeight < 50%
    StorageOpened 0
    call {
        do move moc_para01 10 184
        do talknpc 10 185 c r1
        pause 1
    }
}



########## AUTO GET ##########
automacro AutoGet {
	StorageOpened 1
	CharCurrentWeight < 90%
	timeout 1.5
	# Truyền danh sách item vào đây
	call tryGet "Morrigane's Pendant" "Rosary" "Item3"
}




########## AUTO SELL ##########
automacro AutoSell {
    StorageOpened 1
    CharCurrentWeight > 90%
    call {
        do storage close
        pause 1
        do autosell
    }
}
